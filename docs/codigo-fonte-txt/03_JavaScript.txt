const clamp = (value, min = 0, max = 100) =>
  Math.max(min, Math.min(max, Math.round(value)));

/** Apply a completed self-report check-in without mutating the original student. */
export function applyCheckin(
  student,
  answers,
  { note = "", timestamp = new Date() } = {},
) {
  const ids = quizQuestions.map((question) => question.id);
  if (
    !student ||
    ids.some(
      (id) =>
        !Number.isInteger(answers?.[id]) || answers[id] < 0 || answers[id] > 3,
    )
  ) {
    throw new Error(
      "Complete as três perguntas antes de registrar o check-in.",
    );
  }
  const next = {
    ...student,
    pos: student.pos.map((value) => clamp(value, 10, 98)),
    neg: student.neg.map((value) => clamp(value, 8, 90)),
    history: [...student.history],
    checkins: [...student.checkins],
  };
  const mood = answers.mood,
    energy = answers.energy;
  const moodDelta = [4, 2, 0, -3][mood];
  next.pos[4] = clamp(next.pos[4] + moodDelta, 10, 98);
  next.pos[2] = clamp(next.pos[2] + (mood < 2 ? 1 : -1), 10, 98);
  next.neg[4] = clamp(
    next.neg[4] + (mood === 3 ? 3 : mood === 2 ? 1 : -2),
    8,
    90,
  );
  next.pos[3] = clamp(next.pos[3] + [3, 2, -2, -3][energy], 10, 98);
  next.focus = clamp(next.focus + [2, 1, -2, -3][energy], 20, 98);
  next.history.push(
    clamp(
      next.history.at(-1) +
        Math.round((moodDelta + [3, 2, -2, -3][energy]) / 2),
      20,
      98,
    ),
  );
  if (next.history.length > 8) next.history.shift();
  next.rewards += 25;
  next.checkins.unshift({
    date:
      timestamp instanceof Date
        ? timestamp.toLocaleString("pt-BR", {
            day: "2-digit",
            month: "short",
            hour: "2-digit",
            minute: "2-digit",
          })
        : String(timestamp),
    mood: quizQuestions[0].options[mood][1],
    energy: quizQuestions[1].options[energy][1],
    support: quizQuestions[2].options[answers.support][1],
    note: note.trim() || "Sem observação adicional.",
  });
  return next;
}

const clamp = (value, min, max) => Math.max(min, Math.min(max, value));

/** Arithmetic mean of the student's current illustrative subject grades. */
export function average(student) {
  const grades = student?.grades ?? [];
  if (grades.length === 0) return 0;
  return grades.reduce((sum, [, grade]) => sum + grade, 0) / grades.length;
}

/** Positive share of the configured emotional indicators (demo index, not diagnosis). */
export function wellbeing(student) {
  const positive = student?.pos ?? [];
  const negative = student?.neg ?? [];
  const positiveTotal = positive.reduce((sum, value) => sum + value, 0);
  const total = positiveTotal + negative.reduce((sum, value) => sum + value, 0);
  return total > 0 ? Math.round((positiveTotal / total) * 100) : 0;
}

/** Demo focus indicator; derive from motivation and creativity only if absent. */
export function focusScore(student) {
  if (Number.isFinite(student?.focus))
    return clamp(Math.round(student.focus), 0, 100);
  const values = student?.pos ?? [];
  return values.length > 3 ? Math.round((values[0] + values[3]) / 2) : 0;
}

/** Shared weighted engagement score used by cards, comparison and profile. */
export function engagement(student) {
  const academic = average(student) * 10;
  const attendance = Number.isFinite(student?.attendance)
    ? student.attendance
    : 0;
  return Math.round(
    academic * 0.4 +
      wellbeing(student) * 0.25 +
      attendance * 0.2 +
      focusScore(student) * 0.15,
  );
}

export function familySummary(students) {
  if (!students?.length) return { engagement: 0, wellbeing: 0, attendance: 0 };
  const mean = (values) =>
    Math.round(values.reduce((sum, value) => sum + value, 0) / values.length);
  return {
    engagement: mean(students.map(engagement)),
    wellbeing: mean(students.map(wellbeing)),
    attendance: mean(students.map((student) => student.attendance ?? 0)),
  };
}

const tabPaths = {
  home: "visao-geral",
  emotions: "filhos/emocoes",
  grades: "notas",
  quiz: "quiz-emocional",
  insights: "graficos",
  chat: "mensagens",
};
const pathTabs = Object.fromEntries(
  Object.entries(tabPaths).map(([tab, path]) => [path, tab]),
);
const studentSlugs = { 1: "isabela", 2: "gabriel", 3: "manuela" };
const slugStudents = Object.fromEntries(
  Object.entries(studentSlugs).map(([id, slug]) => [slug, Number(id)]),
);

function cleanBase(base) {
  const segments = String(base || "/")
    .split("/")
    .filter(Boolean);
  return segments.length ? `/${segments.join("/")}/` : "/";
}

export function appUrl(tab, studentId = 1, base = "/") {
  const route = tabPaths[tab] || tabPaths.home;
  const prefix = cleanBase(base);
  const url = `${prefix}${route}`;
  const slug = studentSlugs[studentId];
  return slug && studentId !== 1 ? `${url}?filho=${slug}` : url;
}

export function parseAppUrl(pathname, search = "", base = "/") {
  const prefix = cleanBase(base);
  let relative = String(pathname || "/");
  if (relative.startsWith(prefix)) relative = relative.slice(prefix.length);
  else relative = relative.replace(/^\/+/, "");
  relative = relative.replace(/\/+$/, "");
  const tab = pathTabs[relative] || "home";
  const params = new URLSearchParams(search);
  const studentId = slugStudents[params.get("filho")?.toLowerCase()] || 1;
  return { tab, studentId };
}
